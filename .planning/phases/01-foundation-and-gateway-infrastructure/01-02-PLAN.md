   ```rust
   mod service;

   #[tokio::main]
   async fn main() -> Result<(), Box<dyn std::error::Error>> {
   ```

   ```rust
   pub mod health;

   pub fn create_router(state: AppState) -> Router {
       Router::new()
           .route("/health", get(health::gateway_health))
           .nest("/v1", v1_routes())
           .with_state(state)
   }

   fn v1_routes() -> Router<AppState> {
       Router::new()
           .route("/health/services", get(health::service_health))
   }
   ```

4. Create `services/gateway/src/routes/health.rs`:
   - `gateway_health()` handler: returns `Json({"status": "ok"})` -- no gRPC call, proves gateway is running.
   - `service_health(State(state): State<AppState>)` handler:
     - Generates a UUIDv7 request_id via `uuid::Uuid::now_v7()`
     - Creates `prost_types::Empty {}` requests for each service
     - Inserts `x-request-id` into gRPC metadata on each request: `request.metadata_mut().insert("x-request-id", request_id.parse().unwrap())`
     - Clones each client from state (`state.auth_client.clone()`, etc.)
     - Calls `.health(request)` on each client
     - Builds response JSON:
       ```json
       {
         "status": "ok",
         "request_id": "<uuid>",
         "services": {
           "auth": "ok" | "unavailable",
           "catalog": "ok" | "unavailable",
           "user": "ok" | "unavailable"
         }
       }
       ```
     - Returns `Result<Json<HealthResponse>, AppError>` where gRPC failures map through `AppError::from(tonic::Status)`
     - Note: Individual service failures should NOT fail the whole request. If auth is down but catalog and user are up, return 200 with `"auth": "unavailable"`. Only return an error if ALL services fail or if there is a gateway-level problem.
   - Define `HealthResponse` and `ServiceHealth` structs with `#[derive(Serialize)]`
   - `GatewayHealthResponse` for the simple gateway-only endpoint: `{ "status": "ok" }`
